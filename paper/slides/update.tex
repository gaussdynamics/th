% --- 3. The design changed -----------------------------------------------
\begin{frame}{The Surrogate Design Changed}
  \vspace{0.3em}
  {\small The original plan combined three ideas. One was kept and two were dropped.}

  \vspace{0.6em}
  {\footnotesize
  \begin{tabular}{@{}L{4.2cm}L{1.8cm}L{7.4cm}@{}}
    \toprule
    \textbf{Part of the original plan} & \textbf{Status} & \textbf{Reason} \\
    \midrule
    Graph neural network: vehicles as nodes, couplers as edges
      & \textcolor{chartBlue}{\textbf{Kept}}
      & Handles any train length with one set of weights. \\[0.35em]
    Learn a correction to the physics equations
      & \textcolor{uniCrimson}{\textbf{Dropped}}
      & The training data come from those same equations, so the correction
        is exactly zero. \\[0.35em]
    Neural ODE: learn a derivative and integrate it with a solver
      & \textcolor{uniCrimson}{\textbf{Dropped}}
      & Stiff couplers would force 5 or more network calls per stored step,
        between states no data can check. \\
    \bottomrule
  \end{tabular}}

  \vspace{0.9em}
  \begin{beamercolorbox}[wd=\textwidth,sep=7pt]{block body}
    \small\textbf{New design: a graph network simulator.} The network predicts one
    fixed 0.25\,s step. One network call replaces 50 evaluations of the physics
    equations, and more of the physics is built in than before, not less.
  \end{beamercolorbox}
\end{frame}

% --- 4. Why: the anchor measurement --------------------------------------
\begin{frame}{Why: The Couplers Are Stiff at This Step Size}
  \vspace{0.1em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{tikzpicture}
        \begin{axis}[
            width=\linewidth, height=5.6cm,
            ybar, bar width=10pt, ymode=log, log origin=infty,
            ymin=0.01, ymax=3,
            symbolic x coords={none, euler, frozen, rk4},
            xtick={none, euler, frozen, rk4}, bar shift=0pt,
            xticklabels={{No anchor}, {Euler step}, {RK4, force\\frozen}, {RK4 step}},
            xticklabel style={font=\scriptsize, align=center},
            yticklabel style={font=\scriptsize},
            ylabel={\scriptsize Target Left to Learn (m/s)},
            ymajorgrids, grid style={chartGrid},
            axis x line*=bottom, axis y line*=left,
            enlarge x limits=0.18,
            nodes near coords, nodes near coords style={font=\tiny},
            point meta=explicit symbolic,
          ]
          \addplot[fill=chartRed, draw=none] coordinates {(none,0.0176) [0.018]};
          \addplot[fill=chartBlue, draw=none] coordinates
            {(euler,0.1108) [0.111] (frozen,0.1016) [0.102] (rk4,1.1869) [1.19]};
        \end{axis}
      \end{tikzpicture}
      \srcline{Median over 480 transitions of data/v2. Lower is an easier learning problem.}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      An \textbf{anchor} is a cheap physics estimate of the next state; the
      network would learn only what is left over.

      \vspace{0.5em}
      For velocity, \alert{every physics anchor leaves more to learn}, about
      6$\times$ more at best. One RK4 step of 0.25\,s diverges.

      \vspace{0.5em}
      The cause is stiffness: a coupler between two heavy cars oscillates with
      a period as short as 0.21\,s, and an explicit step of 0.25\,s is unstable
      for it ($\omega h = 7.45$ against a limit of about 2.8).

      \vspace{0.5em}
      For position the anchor helps: $\dot x = v$ exactly, so it is
      \textbf{113$\times$ better}.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{How the GPU Simulator Works}
  \vspace{-0.1em}
  {\footnotesize Same equations as the original simulator. Four changes make it
  fast enough to build the datasets.\par}
  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \gpucard{1. Whole trains at once}{Cost flat in $N$}{%
        \begin{minipage}[t]{0.70\linewidth}
          The original loops over vehicles one at a time in Python. The GPU
          version holds every vehicle of every run in one table (a
          \emph{tensor}) and computes each force for all of them at once.
          Shorter trains are padded with empty vehicles (white).
        \end{minipage}\hfill
        \begin{minipage}[t]{0.27\linewidth}
          \vspace{0.3em}\centering\usebox{\gpugridbox}
        \end{minipage}}

      \vspace{0.35em}
      \gpucard{3. Record one step, replay it}{10$\times$ faster}{%
        Each step was about 660 small GPU jobs, and the GPU sat idle 92\,\%
        of the time waiting for the next one. The step is now recorded once
        (a \emph{CUDA graph}) and replayed as one job: 38\,s down to 3.8\,s.}
    \end{column}
    \begin{column}{0.49\textwidth}
      \gpucard{2. A fixed time step}{0.02\,s step}{%
        The original solver shrinks its step whenever any coupler reaches the
        edge of its slack: on a long train, almost always. The GPU version
        uses a standard fixed-step method (RK4), so every run takes a known
        number of steps and none can stall. States are saved every 0.25\,s.}

      \vspace{0.35em}
      \gpucard{4. Extra digits where needed}{1.2$\times$ the cost}{%
        Calculations keep about 16 digits (\emph{double precision}); results
        are saved with about 7 (\emph{single}). With 7 digits, rounding alone
        used 60\,\% of the allowed speed error. 16 digits cost only 20\,\%
        more time.}
    \end{column}
  \end{columns}

  \vspace{0.4em}
  {\scriptsize
  \textbf{Checked against the original} with 60 tests on 15 reference
  scenarios, written before the GPU version: speed within 0.05\,m/s, couplers
  engage and release at the same moments in $\ge$98\,\% of samples, peak
  coupler force within 5\,\%.\par}
\end{frame}

% --- 9. How the results are measured ---------------------------------------
\begin{frame}{How the Results Are Measured}
  \vspace{0.2em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      {\small\textbf{The test}}
      \begin{itemize}\setlength\itemsep{0.3em}
        \footnotesize
        \item Trained on trains of 10 to 80 cars. Tested on trains of 121 to
              150 cars that it never saw.
        \item \textbf{Rollout:} the model predicts 0.25\,s ahead, then starts
              the next step from its own prediction. 200 steps covers 50\,s.
        \item \textbf{Error:} the worst vehicle (or coupler) in the train,
              compared with the simulator's true trajectory. The typical
              (median) value over many random starting points is reported.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      {\small\textbf{The reference: ``hold''}}
      \begin{itemize}\setlength\itemsep{0.3em}
        \footnotesize
        \item Keep the starting state and never update it.
        \item Its error is therefore \alert{how much the train actually
              changed}. It measures how much there was to predict; it is a
              scale, not a competing model.
        \item Results are stated as \textbf{model error as a share of the
              actual change}: 0\% is perfect, 100\% is no better than
              standing still.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.8em}
  \begin{beamercolorbox}[wd=\textwidth,sep=6pt]{block body}
    \footnotesize Real comparison models (constant-acceleration extrapolation,
    a physics-correction MLP, a transformer) are planned for the evaluation
    chapter and have not been run yet.
  \end{beamercolorbox}
\end{frame}

\begin{frame}{Result: Speed Stays Accurate on Longer Trains}
  \vspace{0.3em}
  \centering
  {\footnotesize
  \begin{tabular}{@{}lccc@{\hspace{2.2em}}ccc@{}}
    \toprule
     & \multicolumn{3}{c}{\textbf{Trained sizes, 10--80 cars}}
     & \multicolumn{3}{c}{\textbf{Unseen sizes, 121--150 cars}} \\
    \cmidrule(r{2.2em}){2-4}\cmidrule{5-7}
    Rollout & Actual change & Model error & Share
            & Actual change & Model error & Share \\
    \midrule
    1 step, 0.25\,s  & 0.016 & 0.004 & 23\%  & 0.028 & 0.011 & 39\% \\
    50 steps, 12.5\,s & 0.48  & 0.028 & 5.7\% & 0.53  & 0.061 & 11\% \\
    200 steps, 50\,s  & 1.78  & 0.066 & \textbf{3.7\%} & 1.31 & 0.118 & \textbf{9.0\%} \\
    \bottomrule
  \end{tabular}}

  \vspace{0.3em}
  \srcline{Speed of the worst vehicle, in m/s. ``Actual change'' is the error of holding the starting state.}

  \vspace{0.8em}
  \begin{minipage}{0.94\textwidth}\small
    \begin{itemize}\setlength\itemsep{0.35em}
      \item After 50\,s, the model's speed error is \textbf{0.066\,m/s} on
            familiar train lengths and \textbf{0.118\,m/s} on trains up to 15
            times longer: under a tenth of how much the speed actually changed.
      \item The one-step share is higher only because a train barely changes
            in a quarter second. The long rollouts are what a simulation uses.
      \item Doubling training time nearly halved the 200-step error, so the
            model is not yet fully trained.
    \end{itemize}
  \end{minipage}
\end{frame}

% --- 12. Why the shape result fails ------------------------------------------
\begin{frame}{Why Coupler Shape Fails: Errors Build Up Instead of Cancelling}
  \vspace{0.1em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      {\small\textbf{Three explanations tested and ruled out}}
      \begin{itemize}\setlength\itemsep{0.3em}
        \footnotesize
        \item \textbf{Too few message-passing rounds?} No: error is even along
              the train, not larger far from the ends.
        \item \textbf{A slow stretch of the whole train?} No: removing it made
              familiar lengths 47\% worse and long trains no better.
        \item \textbf{Caused by the speed error?} No: with perfectly correct
              speeds, only 9\% of the shape error went away.
      \end{itemize}
      \vspace{0.3em}
      {\footnotesize Per step, the model is nearly as accurate on long trains
       (17\% worse). The problem appears only over many steps.}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{tikzpicture}
        \begin{axis}[
            width=\linewidth, height=4.6cm,
            ybar, bar width=14pt, ymin=0, ymax=50,
            symbolic x coords={trained, unseen},
            xtick={trained, unseen},
            xticklabels={{38 cars}, {150 cars}},
            xticklabel style={font=\scriptsize},
            yticklabel style={font=\scriptsize},
            ylabel={\scriptsize Shape Error After 50\,s (mm)},
            ymajorgrids, grid style={chartGrid},
            axis x line*=bottom, axis y line*=left,
            enlarge x limits=0.45,

       accuracy problem.}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \srcline{Right: one diagnostic run on a 38-car and a 150-car train.}
\end{frame}

% --- 13. Open problems ---------------------------------------------------
\begin{frame}{Open Problems}
  \vspace{0.4em}
  \begin{enumerate}\setlength\itemsep{0.7em}
    \small
    \item \textbf{No corridors held out for the control demonstration.} With
          only four corridors, reserving one more leaves too little to train on.
    \item \textbf{The steep-grade test set is built on the rejected corridor},
          so it tests false grades rather than real mountain grades.
    \item \textbf{The national rail network is not yet split into routes.} It is
          a graph of 95,936 segments; turning it into corridors is the next
          piece of code. Everything above waits on it.
    \item \textbf{Coupler shape on long trains} (slides 10 and 11), which matters
          most for the control chapter.
  \end{enumerate}

  \vspace{0.6em}
  \begin{minipage}{0.92\textwidth}\footnotesize\color{uniSlate}
    Planned fix for 1 and 2: extract corridors including real steep crossings
    such as Raton Pass, then rebuild with 50,000 to 100,000 scenarios, about
    2 to 4 hours on the GPU. Results so far will be re-run on the new data.
  \end{minipage}
\end{frame}
