\eqslide{1}{Stage 1 Equations: Forces on a Single Vehicle}{%
  \vspace{-0.2em}
  \[
    \eqterm{eqInert}{m\,\dot v}{mass $\times$\\acceleration}
    = \eqnew{eqTrac}{F_{\mathrm{trac}}(t)}{traction\\(pulls forward)}
    - \eqnew{eqBrk}{F_{\mathrm{brk}}(t)}{braking}
    - \eqnew{eqRes}{R(v)}{rolling and\\air resistance}
    - \eqnew{eqGrade}{G(x)}{grade\\(slope)}
    - \eqnew{eqCurv}{C_{\kappa}(x,v)}{curve\\resistance}
    \qquad\qquad \dot x = v
  \]
  \vspace{-0.3em}
  \eqcards
    {\eqcard{eqRes}{Davis resistance}{%
      $R(v) = \bigl(A + B|v| + C v^2\bigr)\,\mathrm{sgn}(v)$\\[2pt]
      $A$: rolling friction. $B$: losses that grow with speed. $C$: air drag.
      The sign factor makes it always oppose the motion.\\[1pt]
      Locomotive: $A = 800$\,N, $B = 15$, $C = 0.8$ (SI units).}}
    {\eqcard{eqGrade}{Grade and curvature}{%
      {\color{eqGrade}$G(x) = m g \sin\theta(x)$}, with $\theta$ the slope
      angle of the track. Positive when climbing.\\[2pt]
      {\color{eqCurv}$C_{\kappa} = c_{\kappa}\, m g\, \lambda\,|\kappa(x)|\,\mathrm{sgn}(v)$},
      with $\kappa = 1/\text{radius}$ the curvature of the track.}}
    {\eqcard{eqTrac}{Traction command}{%
      $F_{\mathrm{trac}}(t)$ is a given function of time: a smooth ramp from
      0 to 180\,kN between $t = 5$ and $20$\,s.\\[2pt]
      A ramp, not a step, so the force has no sudden jump for the solver to
      handle.}}
  \eqfoot{State: position and speed $(x, v)$, two numbers. The test run uses
    level track, no braking and no curves, so only traction and Davis
    resistance act. Tracker above: solid = added at this stage, tinted =
    added earlier, outlined = added later.}
}

\eqslide{2}{Stage 2 Equations: Two Vehicles Joined by a Spring Coupler}{%
  \vspace{-0.4em}
  \[
    \renewcommand{\arraystretch}{1.0}
    \begin{array}{r@{\;=\;}l@{\;}l@{\;}l@{\;}l@{\;}l}
      \eqc{eqInert}{m_0\dot v_0}
        & \eqc{eqTrac}{F_{\mathrm{trac}}}
        & -\,\eqc{eqBrk}{F_{\mathrm{brk},0}}
        & -\,\eqc{eqRes}{R_0(v_0)}
        & -\,\eqc{eqGrade}{G_0(x_0)}
        & \eqnew{eqCpl}{-\,F}{coupler holds\\the locomotive back} \\
      \eqc{eqInert}{m_1\dot v_1}
        & \hphantom{F_{\mathrm{trac}}}
        & -\,\eqc{eqBrk}{F_{\mathrm{brk},1}}
        & -\,\eqc{eqRes}{R_1(v_1)}
        & -\,\eqc{eqGrade}{G_1(x_1)}
        & \eqnew{eqCpl}{+\,F}{coupler pulls\\the car forward}
    \end{array}
  \]
  \vspace{-0.4em}
  \eqcards
    {\eqcard{eqCpl}{Coupler stretch}{%
      $\delta = (x_0 - x_1) - L_0, \qquad \dot\delta = v_0 - v_1$\\[2pt]
      $L_0 = 20$\,m is the relaxed spacing. $\delta > 0$: stretched, called
      \emph{draft}. $\delta < 0$: compressed, called \emph{buff}.}}
    {\eqcard{eqCpl}{Linear spring-damper}{%
      $F = k\,\delta + c\,\dot\delta$\\[2pt]
      Stiffness $k = 8$\,MN/m: force per metre of stretch. Damping
      $c = 0.4$\,MN$\cdot$s/m: force per m/s of stretching speed, which
      removes energy from oscillations.}}
    {\eqcard{eqInert}{Equal and opposite}{%
      The same $F$ appears with opposite signs in the two equations (Newton's
      third law), so the coupler moves force between the vehicles but adds
      none to the train.}}
  \eqfoot{State: $(x_0, x_1, v_0, v_1)$, four numbers. Colours match the
    Stage~1 terms; the car has no traction.}
}

\eqslide{3}{Stage 3 Equations: A Coupler with Slack}{%
  {\small The two equations of motion from Stage~2 do not change. Only the
  coupler law is replaced: the linear
  {\color{uniSlate}$F = k\delta + c\dot\delta$} becomes\par}
  \vspace{-0.4em}
  \[
    \eqc{eqCpl}{F(\delta,\dot\delta)} =
    \left\{\;
    \renewcommand{\arraystretch}{1.35}
    \begin{array}{lll}
      \eqcnew{eqCpl}{k_d(\delta - s) + c_d\,\dot\delta}
        & \delta > s
        & \text{\small\color{eqCpl} draft: stretched past the slack} \\
      \eqcnew{uniSlate}{0}
        & |\delta| \le s
        & \text{\small\color{uniSlate} slack: no contact, no force} \\
      \eqcnew{eqCpl}{k_b(\delta + s) + c_b\,\dot\delta}
        & \delta < -s
        & \text{\small\color{eqCpl} buff: compressed past the slack}
    \end{array}\right.
  \]
  \vspace{-0.2em}
  \eqcards
    {\eqcard{uniSlate}{Slack}{%
      $s = 20$\,mm of free play each way in the coupler and draft gear.
      Inside it the vehicles move independently.}}
    {\eqcard{eqCpl}{Different stiffness each way}{%
      Draft $k_d = 9$\,MN/m; buff $k_b = 12$\,MN/m.\\[2pt]
      Stiffer when pushed together than when pulled apart. Only the travel
      past the slack is loaded.}}
    {\eqcard{uniCrimson}{Why this matters}{%
      The law has corners at $\delta = \pm s$. When slack closes the force
      rises from zero very quickly, so the solver must take very small time
      steps near these points.}}
  \eqfoot{A simplified draft gear model. Hysteresis (different force when
    loading and unloading) is not modelled.}
}

\eqslide{4}{Stage 4 Equations: Force Passed Along a Chain of Couplers}{%
  \vspace{-0.5em}
  \[
    \renewcommand{\arraystretch}{1.05}
    \begin{array}{r@{\;=\;}l@{\;}l@{\;}l@{\;}l@{\;}l}
      \eqc{eqInert}{m_0\dot v_0}
        & \eqc{eqTrac}{F_{\mathrm{trac}}}
        & -\,\eqc{eqRes}{R_0}
        & -\,\eqc{eqGrade}{G_0}
        &
        & \eqcnew{eqCpl}{-\,F_0} \\
      \eqc{eqInert}{m_1\dot v_1}
        & \hphantom{F_{\mathrm{trac}}}
        & -\,\eqc{eqRes}{R_1}
        & -\,\eqc{eqGrade}{G_1}
        & \eqnew{eqCpl}{+\,F_0}{pulled forward by\\the coupler ahead}
        & \eqnew{eqCpl}{-\,F_1}{held back by\\the coupler behind} \\
      \eqc{eqInert}{m_2\dot v_2}
        & \hphantom{F_{\mathrm{trac}}}
        & -\,\eqc{eqRes}{R_2}
        & -\,\eqc{eqGrade}{G_2}
        & \eqcnew{eqCpl}{+\,F_1}
        &
    \end{array}
  \]
  \vspace{-0.5em}
  \eqcards
    {\eqcard{eqCpl}{One law per coupler}{%
      Coupler $j$ joins vehicles $j$ and $j+1$:\\
      $\delta_j = (x_j - x_{j+1}) - L_{0,j}$\\
      $F_j = F(\delta_j, \dot\delta_j)$, the Stage~3 slack law.}}
    {\eqcard{eqCpl}{Front in, rear out}{%
      A middle vehicle is pulled by the coupler ahead and held back by the one
      behind. The rear car feels the locomotive only once the slack in every
      coupler ahead of it has closed.}}
    {\eqcard{eqBrk}{Stage 4b: brake on the lead only}{%
      Adds $-\,F_{\mathrm{brk},0}\tanh(v_0/\varepsilon)$ to the lead:
      80\,kN from 45 to 75\,s, traction still on. Coupler forces drop (lead
      coupler 109 to 41\,kN) but stay in draft. $\tanh$: see Stage~7.}}
  \eqfoot{State: $(x_0, x_1, x_2, v_0, v_1, v_2)$, six numbers. Curvature is
    off; braking is off except in the 4b run.}
}

\eqslide{6}{Stage 6 Equations: Each Vehicle Reads the Route at Its Own Position}{%
  \vspace{-0.5em}
  \[
    \eqc{eqInert}{m_i\dot v_i}
    = \eqc{eqTrac}{F_{\mathrm{trac},i}}
    - \eqc{eqBrk}{F_{\mathrm{brk},i}}
    - \eqc{eqRes}{R_i(v_i)}
    - \eqnew{eqGrade}{m_i g \sin\theta(x_i)}{grade where\\vehicle $i$ is}
    - \eqnew{eqCurv}{c_{\kappa}\, m_i g\, \lambda\, |\kappa(x_i)|\,\mathrm{sgn}(v_i)}{curve resistance\\where vehicle $i$ is}
    \eqc{eqCpl}{\;+\,F_{i-1} - F_i}
  \]
  \vspace{-0.4em}
  \eqcards
    {\eqcard{eqInert}{The route as data}{%
      Slope $\sin\theta(s)$ and curvature $\kappa(s)$ are stored at points
      along the track and linearly interpolated in between. Before this
      stage, one value applied to the whole train; now each vehicle looks up
      its own $x_i$.}}
    {\eqcard{eqGrade}{Why it matters}{%
      A long freight train can have its head on a climb while its tail is still on
      flat track. The grade force then differs along the train, so couplers
      stretch and compress even at a constant throttle setting.}}
    {\eqcard{eqCurv}{Curve resistance}{%
      A fixed fraction of vehicle weight, set by the curve radius and not by
      speed. $\lambda \approx 0.699$ is the AREMA value of 0.8\,lb per ton
      per degree of curve. $c_{\kappa} = 1$ is the standard value;
      $c_{\kappa} = 0$ turns the term off.}}
  \eqfoot{Test run: slope rises from level to 0.8\,\% between 2 and 6\,km,
    with an 800\,m radius curve there. This notebook run used an older curve
    term that grew with $v^2$; all later results use the form above.}
}

\eqslide{7}{Stage 7 Equations: Commands Act Through Actuator Lag}{%
  \vspace{-0.6em}
  \[
    \eqc{eqInert}{m_i\dot v_i}
    = \eqnew{eqTrac}{F_{\mathrm{trac},i}(z_{\mathrm{trac},i},\,v_i)}{traction from its lag\\state, power limited}
    - \eqnew{eqBrk}{F_{\mathrm{brk},i}(z_{\mathrm{brk},i})\,\tanh(v_i/\varepsilon_b)}{brake from its lag state,\\opposing the motion}
    - \eqc{eqRes}{R_i}
    - \eqc{eqGrade}{G_i(x_i)}
    - \eqc{eqCurv}{C_i(x_i)}
    \eqc{eqCpl}{\;+\,F_{i-1} - F_i}
  \]
  \vspace{-0.9em}
  \[
    \eqnew{eqAct}{\dot z_{\mathrm{trac},i} = \frac{u_{\mathrm{trac},i} - z_{\mathrm{trac},i}}{\tau_t}}{traction lag}
    \qquad
    \eqnew{eqAct}{\dot z_{\mathrm{brk},i} = \frac{u_{\mathrm{brk},i} - z_{\mathrm{brk},i}}{\tau_b}}{brake lag}
  \]
  \vspace{-0.6em}
  \eqcards
    {\eqcard{eqAct}{Actuator lag}{%
      $u$ is the command; $z$ is what the equipment delivers. $z$ covers
      63\,\% of a change in $u$ within one time constant $\tau$
      ($\tau_t = 5$\,s, $\tau_b = 3$\,s).}}
    {\eqcard{eqAct}{Power limit}{%
      $F_{\mathrm{trac},i} = \min\bigl(z_{\mathrm{trac},i},\; P_{\max}/|v_i|\bigr)$\\[2pt]
      Force $\times$ speed cannot exceed $P_{\max} = 3.5$\,MW: at most
      175\,kN at 20\,m/s. ($z \ge 0$; $|v_i|$ floored at 1\,m/s.)}}
    {\eqcard{eqBrk}{Brake direction}{%
      $F_{\mathrm{brk},i} = \min\bigl(z_{\mathrm{brk},i},\; F_{\mathrm{brk},\max,i}\bigr)$\\[2pt]
      $\tanh(v/\varepsilon_b)$ is $+1$ moving forward, $-1$ backward, and
      smooth through 0 at a stop (see the next two slides).}}
  \eqfoot{State: $x$, $v$, $z_{\mathrm{trac}}$, $z_{\mathrm{brk}}$ per
    vehicle, $4N$ numbers. Force follows a command through $z$, never

      \onslide<2->{%
      \centering
      {\footnotesize\color{chartBlue!70!black}\textbf{Corrected}}

      \vspace{0.2em}
      \includegraphics[width=\linewidth,height=0.52\textheight,keepaspectratio]%
        {../../notebooks/notebook_images/ltd_stage07_brake_buildup_power_limited_traction_states.png}}

  \vspace{0.3em}
  \begin{minipage}{0.96\textwidth}\footnotesize
    \only<1>{Brake force always pushed in the $-x$ direction, regardless of
      which way the train was moving. The speed passes through zero and keeps
      falling to \alert{$-99$\,km/h}: a train moving backward at 99\,km/h under
      full braking.}
    \only<2>{The fix makes brake force oppose the direction of motion, using a
      smooth $\tanh(v/\varepsilon)$ blend instead of $\mathrm{sgn}(v)$. The error
      went unnoticed because all earlier scenarios started from rest under
      traction. \textbf{Randomized starting conditions exposed it.} The fix
      changed \alert{only one} of the nine stage figures.}
  \end{minipage}
\end{frame}
